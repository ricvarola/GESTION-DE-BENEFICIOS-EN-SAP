const getWorkingDays = (startDate, endDate) => {
  if (!startDate || !endDate) return 0;
  let count = 0;
  try {
    let curDate = new Date(startDate);
    const end = new Date(endDate);
    if (isNaN(curDate.getTime()) || isNaN(end.getTime())) return 0;
    while (curDate <= end) {
      const dayOfWeek = curDate.getDay();
      if (dayOfWeek !== 0 && dayOfWeek !== 6) count++;
      curDate.setDate(curDate.getDate() + 1);
    }
  } catch (e) { return 0; }
  return count;
};

const formatDateUI = (dateStr) => {
    if (!dateStr) return null;
    const [y, m, d] = dateStr.split('-');
    return `${d}/${m}/${y}`;
};

const calculateDynamicGap = (estimatedDateStr, actualCloseStr) => {
  if (!estimatedDateStr || estimatedDateStr === 'N/A') return null;
  const parseDate = (s) => {
    if (!s) return null;
    const date = new Date(s);
    return isNaN(date.getTime()) ? null : date;
  };
  const estDate = parseDate(estimatedDateStr);
  if (!estDate) return null;
  let comparisonDate;
  const realDate = parseDate(actualCloseStr);
  if (realDate) { comparisonDate = realDate; } 
  else { comparisonDate = new Date(); }
  comparisonDate.setHours(0, 0, 0, 0);
  estDate.setHours(0, 0, 0, 0);
  const diffTime = comparisonDate.getTime() - estDate.getTime();
  return Math.ceil(diffTime / (1000 * 60 * 60 * 24));
};

                 <button onClick={handleApplyChanges} className="bg-green-600 text-white px-8 py-3 rounded-2xl text-[10px] font-black flex items-center gap-2 uppercase tracking-widest"><SaveAll size={16}/> Guardar Cambios</button>
               </div>
             </div>
             
             <div className="p-8 grid grid-cols-1 md:grid-cols-2 gap-6 bg-white border-b border-slate-100">
                <div className="space-y-1"><label className="text-[10px] font-black text-slate-400 uppercase tracking-widest">Título</label><input type="text" value={tempProjectData.name} onChange={(e) => setTempProjectData({...tempProjectData, name: e.target.value})} className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2 text-sm font-bold text-slate-700" /></div>
             </div>

             <div className="overflow-x-auto p-4">
               <table className="w-full text-left min-w-[800px]">
                 <thead><tr className="bg-white text-[9px] font-black text-slate-400 uppercase tracking-widest border-b border-slate-100"><th className="px-4 py-5 w-72">Entregable</th><th className="px-4 py-5 w-20 text-center">Avance %</th></tr></thead>
                 <tbody className="divide-y divide-slate-50">
                   {tempProjectData.macros.map((macro) => (
                     <Fragment key={macro.id}>
                       <tr className="bg-slate-50"><td colSpan="2" className="px-4 py-4 font-black uppercase text-xs text-slate-800">{macro.id} - {macro.name}</td></tr>
                       {macro.deliverables.map((item) => (
                         <tr key={item.id} className="hover:bg-blue-50/20">
                           <td className="px-4 py-4"><input type="text" value={item.name} onChange={(e) => handleTempItemChange(macro.id, item.id, 'name', e.target.value)} className="bg-transparent border-none w-full focus:outline-none text-xs font-bold text-slate-600 uppercase" /></td>
                           <td className="px-4 py-4 text-center"><input type="number" min="0" max="100" value={item.progress} onChange={(e) => handleTempItemChange(macro.id, item.id, 'progress', e.target.value)} className="w-16 bg-white border border-slate-200 rounded-lg py-1 text-center text-xs font-black text-blue-600 outline-none"/></td>
                         </tr>
                       ))}
                     </Fragment>
                   ))}
                 </tbody>
               </table>
             </div>
          </div>
        </main>
      )}
    </div>
  );
};
